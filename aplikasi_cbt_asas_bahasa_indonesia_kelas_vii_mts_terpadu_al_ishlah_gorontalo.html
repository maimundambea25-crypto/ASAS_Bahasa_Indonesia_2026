☁️ Panduan Pemasangan Google Apps Script (Code.gs)

  1. Buka Google Spreadsheet baru, lalu klik menu Ekstensi > Apps Script.
  2. Hapus kode bawaan dan tempel kode di bawah ini, lalu klik Terapkan (Deploy) > Deployment Baru > Aplikasi Web.
  3. Pilih Yang memiliki akses: Siapa saja (Anyone), lalu salin URL Web App ke Panel Guru.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "ID Bukti", "Nama Siswa", "Kelas", "Nilai Akhir", "Skor PG", "Skor BS", "Status Uraian", "Pelanggaran"]);
  }
  sheet.appendRow([
    new Date(), data.idBukti, data.nama, data.kelas, data.skorAkhir,
    data.detailPG, data.detailBS, data.detailUraian, data.pelanggaran
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
