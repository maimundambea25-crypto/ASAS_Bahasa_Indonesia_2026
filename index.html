function getOrCreateSheet(name, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let s = ss.getSheetByName(name);
  if (!s) { s = ss.insertSheet(name); s.appendRow(headers); s.getRange(1,1,1,headers.length).setFontWeight('bold'); }
  return s;
}
function getLksSheet() { return getOrCreateSheet(SHEETS.LKS, ['Timestamp','ID Bukti','Nama Siswa','Kelas','Nilai Objektif','Skor PG','Skor BS','Status Uraian','Pelanggaran','Waktu Selesai','Payload_LKS_JSON']); }
function getLiveSheet() { return getOrCreateSheet(SHEETS.LIVE, ['UpdatedAt','StudentKey','ID Sesi','Nama Siswa','Kelas','Status','Terjawab','Total Soal','Soal Aktif','Live Skor','Poin PG','Poin BS','Sisa Detik','Pelanggaran','Alasan Pelanggaran Terakhir','Payload_JSON']); }
function getViolationSheet() { return getOrCreateSheet(SHEETS.VIOLATION, ['Timestamp','StudentKey','Nama Siswa','Kelas','Jumlah Pelanggaran','Status','Alasan','ID Sesi','Aktif']); }
function getUnblockSheet() { return getOrCreateSheet(SHEETS.UNBLOCK, ['UnblockedAt','StudentKey','Nama Siswa','Kelas','Diizinkan Oleh','EpochMs']); }
function makeStudentKey(k, n) { return String(k||'').trim().toLowerCase() + '__' + String(n||'').trim().toLowerCase(); }
function buildOutput(obj, cb) {
  const txt = JSON.stringify(obj);
  if (cb && /^[a-zA-Z0-9_$.]+$/.test(cb)) return ContentService.createTextOutput(cb + '(' + txt + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  return ContentService.createTextOutput(txt).setMimeType(ContentService.MimeType.JSON);
}
function upsertLiveRow(data) {
  const sheet = getLiveSheet(), sKey = makeStudentKey(data.kelas, data.nama), lastRow = sheet.getLastRow();
  let targetRow = -1;
  if (lastRow > 1) {
    const keys = sheet.getRange(2, 2, lastRow - 1, 1).getValues();
    for (let i = 0; i < keys.length; i++) { if (String(keys[i][0]) === sKey) { targetRow = i + 2; break; } }
  }
  const row = [new Date(), sKey, data.idBukti||'', data.nama||'', data.kelas||'', data.status||'Sedang Ujian', data.answeredCount??0, data.totalQuestions??25, data.currentQuestionNo??1, data.liveScore??0, data.skorPG??0, data.skorBS??0, data.remainingSeconds??0, data.violations??0, data.lastViolationReason||'', JSON.stringify(Object.assign({}, data, { studentKey: sKey, updatedAtIso: new Date().toISOString() }))];
  if (targetRow > 0) sheet.getRange(targetRow, 1, 1, row.length).setValues([row]); else sheet.appendRow(row);
}

function doGet(e) {
  const p = (e && e.parameter) ? e.parameter : {}, action = String(p.action || 'get_all').trim(), cb = p.callback ? String(p.callback).trim() : '';
  if (action === 'ping') return buildOutput({ status: 'ok', teacher: TEACHER_NAME }, cb);
  if (action === 'check_student_status') {
    const sKey = makeStudentKey(p.kelas, p.nama);
    let latestUnblockMs = 0, isDisq = false, activeViol = 0;
    const uS = getUnblockSheet(), uL = uS.getLastRow();
    if (uL > 1) { const uD = uS.getRange(2,2,uL-1,5).getValues(); for (let i=0;i<uD.length;i++) if (String(uD[i][0])===sKey) latestUnblockMs = Math.max(latestUnblockMs, Number(uD[i][4]||0)); }
    const vS = getViolationSheet(), vL = vS.getLastRow();
    if (vL > 1) { const vD = vS.getRange(2,2,vL-1,8).getValues(); for (let i=0;i<vD.length;i++) if (String(vD[i][0])===sKey && String(vD[i][7])==='AKTIF') { activeViol = Math.max(activeViol, Number(vD[i][3]||0)); if (String(vD[i][4])==='Terdiskualifikasi' || activeViol>=3) isDisq = true; } }
    return buildOutput({ status: 'success', isDisqualified: isDisq, activeViolations: activeViol, latestUnblockMs: latestUnblockMs }, cb);
  }
  if (!TEACHER_PINS.includes(String(p.pin||'').trim())) return buildOutput({ status: 'unauthorized', records: [], liveSessions: [], violations: [] }, cb);
  const records = [], liveSessions = [], violations = [];
  const lksS = getLksSheet(), lksL = lksS.getLastRow();
  if (lksL > 1) { const col = lksS.getRange(2,11,lksL-1,1).getValues(); for (let i=col.length-1;i>=0;i--) { try { if (col[i][0]) records.push(JSON.parse(col[i][0])); } catch(e){} } }

  if (lvL > 1) { const col = lvS.getRange(2,16,lvL-1,1).getValues(); for (let i=col.length-1;i>=0;i--) { try { if (col[i][0]) liveSessions.push(JSON.parse(col[i][0])); } catch(e){} } }

}
